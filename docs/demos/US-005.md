          TYPECHECK: ${{ steps.setup.outputs.cmd_typecheck }}
          LINT: ${{ steps.setup.outputs.cmd_lint }}
          BUILD: ${{ steps.setup.outputs.cmd_build }}
          # setup-project's "Start local services" step already ran the manifest's
          # `services` command (`npx supabase start`) before this step, so TEST runs
